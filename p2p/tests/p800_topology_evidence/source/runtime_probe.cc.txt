#include <cuda.h>
#include <cuda_runtime.h>
#include <cstdio>

int main() {
  int runtime = -1, driver = -1, count = -1;
  auto rr = cudaRuntimeGetVersion(&runtime);
  auto dr = cuDriverGetVersion(&driver);
  auto nr = cudaGetDeviceCount(&count);
  std::printf("cudaRuntimeGetVersion rc=%d value=%d\n", int(rr), runtime);
  std::printf("cuDriverGetVersion rc=%d value=%d\n", int(dr), driver);
  std::printf("cudaGetDeviceCount rc=%d count=%d\n", int(nr), count);
  if (nr != cudaSuccess) return 1;
  for (int i = 0; i < count; ++i) {
    char bdf[64] = {};
    auto rc = cudaDeviceGetPCIBusId(bdf, sizeof(bdf), i);
    std::printf("ordinal=%d bdf_rc=%d bdf=%s\n", i, int(rc), bdf);
    if (rc != cudaSuccess) return 2;
  }
  return 0;
}
