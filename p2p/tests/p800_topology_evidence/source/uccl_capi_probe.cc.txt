#include "uccl_engine.h"
#include <cstdio>
#include <vector>

int main() {
  std::setbuf(stdout, nullptr);
  std::puts("create_begin");
  auto* engine = uccl_engine_create(1, false);
  if (!engine) return 1;
  std::puts("create_pass");
  std::vector<unsigned char> buffer(1024 * 1024, 0x37);
  uccl_mr_t mr = 0;
  int rc = uccl_engine_reg(engine, reinterpret_cast<uintptr_t>(buffer.data()),
                           buffer.size(), mr);
  std::printf("cpu_register_rc=%d\n", rc);
  if (rc != 0) { uccl_engine_destroy(engine); return 2; }
  char* metadata = nullptr;
  rc = uccl_engine_get_metadata(engine, &metadata);
  std::printf("metadata_rc=%d metadata=%s\n", rc, metadata ? metadata : "null");
  delete[] metadata;
  uccl_engine_mr_destroy(engine, mr);
  uccl_engine_destroy(engine);
  std::puts("cleanup_pass");
  return rc == 0 ? 0 : 3;
}
